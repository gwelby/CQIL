class HeartFieldCommunicationsProtocol {
  constructor(heartField) {
    this.heartField = heartField;
    this.messageQueue = [];
    this.subscribers = new Map();
    this.coherenceRequired = 0.6; // Minimum coherence for message delivery
  }
  
  // Send a quantum message through the Heart Field
  send(sourceId, targetId, message, priority = 'normal') {
    const qMessage = {
      sourceId,
      targetId,
      content: message,
      priority,
      timestamp: Date.now(),
      coherence: 0,
      delivered: false
    };
    
    // Add to message queue
    this.messageQueue.push(qMessage);
    
    // Process the queue
    this._processQueue();
    
    return qMessage;
  }
  
  // Subscribe a component to receive messages
  subscribe(componentId, callback) {
    this.subscribers.set(componentId, callback);
  }
  
  // Process the message queue
  _processQueue() {
    // Sort by priority and timestamp
    this.messageQueue.sort((a, b) => {
      if (a.priority === 'high' && b.priority !== 'high') return -1;
      if (a.priority !== 'high' && b.priority === 'high') return 1;
      return a.timestamp - b.timestamp;
    });
    
    // Process messages
    for (let i = 0; i < this.messageQueue.length; i++) {
      const message = this.messageQueue[i];
      
      // Skip already delivered messages
      if (message.delivered) continue;
      
      // Get the connection between source and target
      const connectionId = `${message.sourceId}:${message.targetId}`;
      const connection = this.heartField.connections.get(connectionId);
      
      if (connection) {
        // Update message coherence based on connection
        message.coherence = connection.coherence;
        
        // If coherence is sufficient, deliver the message
        if (message.coherence >= this.coherenceRequired) {
          const targetCallback = this.subscribers.get(message.targetId);
          if (targetCallback) {
            // Deliver the message
            targetCallback(message);
            message.delivered = true;
          }
        }
      }
    }
    
    // Remove delivered messages
    this.messageQueue = this.messageQueue.filter(msg => !msg.delivered);
  }
}
